Relationship source and target
================================================================================

/**
 * KNOWS
 *
 * A person knows another person.
 *
 * @label KNOWS
 * @source 1..* Each :KNOWS may start at one or more people.
 * @target 0..1 Each :KNOWS ends at
 *   at most one person.
 */

--------------------------------------------------------------------------------

(document
  (name)
  (description
    (description_line))
  (label_tag
    name: (identifier))
  (source_tag
    cardinality: (cardinality)
    description: (tag_description
      (description_line)))
  (target_tag
    cardinality: (cardinality)
    description: (tag_description
      (description_line)
      (description_line))))
